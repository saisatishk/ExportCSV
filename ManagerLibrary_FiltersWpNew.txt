<script>
/* Toggle an accordion row open / closed */
function fpToggle(btn) {
  var group = btn.closest('.filter-group');
  group.classList.toggle('fp-collapsed');
}

/* Clear all: reset every PnP combobox and date-range input */
function fpClearAll() {
  var panel = document.querySelector('.filter-panel');
  if (!panel) return;

  /* Reset native selects inside pnp-filtercombobox */
  panel.querySelectorAll('pnp-filtercombobox').forEach(function(el) {
    el.querySelectorAll('option').forEach(function(o) {
      o.removeAttribute('data-selected');
      o.selected = false;
    });
    /* Try to fire the combobox's own clear if it exposes one */
    if (typeof el.clearFilters === 'function') el.clearFilters();
  });

  /* Reset date-range inputs */
  panel.querySelectorAll('pnp-filterdaterange input[type="text"], pnp-filterdaterange input[type="date"]')
    .forEach(function(i) { i.value = ''; });

  /* Hide the active-count badge */
  var badge = document.getElementById('fp-active-count');
  if (badge) { badge.textContent = '0'; badge.classList.remove('fp-visible'); }

  /* Collapse all rows */
  panel.querySelectorAll('.filter-group').forEach(function(g) {
    g.classList.add('fp-collapsed');
  });
}
</script>
